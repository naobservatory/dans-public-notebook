Model

simulate_epidemic <- function(
  population_size,
  doubling_time,
  shedding_duration,
  day_zero
){
  growth_rate <- log(2) / doubling_time
  # Grow until ~10% of people have been infected.
  # With growth rate r, cumulative incidence is approximately (1/r) exp(r t).
  # CI \approx N / 10 => t \approx log(N r / 10) / r
  max_day <- ceiling(log(growth_rate * population_size / 10) / growth_rate)
  tibble(
    day = 0:max_day,
    # Exponential growth from a single case
    incidence = exp(day * growth_rate) / population_size,
    cumulative_incidence = cumsum(incidence),
    # Assume constant shedding for `shedding_duration` days
    prevalence = slide_dbl(incidence, sum, .before = shedding_duration - 1, .complete = FALSE),
    date = day_zero + day,
  )
}
calculate_read_depth <- function(input_params){
  input_params %>%
    mutate(
      samples_per_year = 365 / sampling_interval * num_sites,
      runs_per_year = (annual_budget - num_sites * cost_per_site_per_year - samples_per_year * cost_per_sample) / cost_per_run,
      reads_per_year = reads_per_run * runs_per_year,
      reads_per_sample = reads_per_year / samples_per_year
    )
}
simulate_detection <- function(epi_sim,
                             sampling_interval, 
                             num_sites, 
                             ra_p_01, 
                             phi, 
                             detection_reads, 
                             delay_days,
                             annual_budget, 
                             cost_per_site_per_year, 
                             cost_per_sample, 
                             cost_per_run, 
                             reads_per_run,
                             reads_per_sample,
                             ... # absorb unused parameters
                             ) {
  
  # Run simulation
  mu <- ra_p_01 * 100
  num_reps <- 100
  epi_sim %>%
    # Get epi model on sampling days
    filter(day %% sampling_interval == 0) %>%
    # Data per site and per rep
    cross_join(
      expand_grid(rep = 1:num_reps, site = 1:num_sites)
    ) %>%
    # Simulate reads
    mutate(
      expected_reads = prevalence * mu * reads_per_sample,
      reads = rnbinom(n(), size = phi, mu = expected_reads),
    ) %>%
    # Add sites together
    summarise(
      expected_reads = sum(expected_reads),
      reads = sum(reads),
      .by = day:rep,
    ) %>%
    # Get cumulative reads
    mutate(
      expected_cumulative_reads = cumsum(expected_reads),
      cumulative_reads = cumsum(reads),
      .by = rep,
    ) %>%
    # Simulate detection
    summarise(
      threshold_day = day[which.max(cumulative_reads >= detection_reads)],
      detection_day = threshold_day + delay_days,
      .by = rep,
    ) %>%
    left_join(
      epi_sim,
      by = join_by(detection_day == day),
    ) %>%
    # Return median of reps 
    summarise(
      across(c(date, cumulative_incidence), median),
    )
}
epi_params <- expand_grid(
  population_size = 340e6, # US population
  doubling_time = c(2, 3),
  shedding_duration = 7,
  day_zero = ymd(20200101), # Arbitrary
) %>%
  mutate(epi_sim = pmap(., simulate_epidemic))


sensitivity_params <- expand_grid(
  ra_p_01 = 3e-7, # 15th percentile of SARS-CoV-2 from swab-based P2RA
  phi = 1, # prior mode for wastewater phi in the swab p2ra blog post
  detection_reads = 5,
)

budget_params <- expand_grid(
  annual_budget = 10e6,
  cost_per_site_per_year = 1e5, # completely made up
  cost_per_sample = 1e3, # completely made up
)

sequencing_params <- tribble(
  ~reads_per_run, ~cost_per_run, ~delay_days,
  10e9, 12e3, 5, # 10B flow cell
  25e9, 16e3, 6, # 25B flow cell
)

sampling_params <- expand_grid(
  sampling_interval = 3,
  num_sites = 10,
) %>%
  cross_join(budget_params) %>%
  cross_join(sequencing_params) %>%
  calculate_read_depth

params <- epi_params %>%
  cross_join(sampling_params) %>%
  cross_join(sensitivity_params)

params %>%
  mutate(simulation = pmap(., simulate_detection)) %>%
  unnest(simulation)
